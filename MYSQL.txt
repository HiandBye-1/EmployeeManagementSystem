-- =========================================
-- CREATE DATABASE
-- =========================================
DROP DATABASE IF EXISTS employeeData;
CREATE DATABASE employeeData;
USE employeeData;

-- =========================================
-- EMPLOYEES
-- =========================================
DROP TABLE IF EXISTS employees;
CREATE TABLE employees (
  empid INT NOT NULL AUTO_INCREMENT,
  Fname VARCHAR(65) NOT NULL,
  Lname VARCHAR(65) NOT NULL,
  email VARCHAR(65) NOT NULL,
  HireDate DATE,
  Salary DECIMAL(10,2) NOT NULL,
  SSN VARCHAR(12),
  addressID INT NOT NULL,
  PRIMARY KEY (empid)
);

-- =========================================
-- PAYROLL
-- =========================================
DROP TABLE IF EXISTS payroll;
CREATE TABLE payroll (
  payID INT NOT NULL AUTO_INCREMENT,
  pay_date DATE,
  earnings DECIMAL(8,2),
  fed_tax DECIMAL(7,2),
  fed_med DECIMAL(7,2),
  fed_SS DECIMAL(7,2),
  state_tax DECIMAL(7,2),
  retire_401k DECIMAL(7,2),
  health_care DECIMAL(7,2),
  empid INT,
  PRIMARY KEY (payID)
);

-- =========================================
-- EMPLOYEE JOB TITLES
-- =========================================
DROP TABLE IF EXISTS employee_job_titles;
CREATE TABLE employee_job_titles (
  empid INT NOT NULL,
  job_title_id INT NOT NULL
);

-- =========================================
-- JOB TITLES
-- =========================================
DROP TABLE IF EXISTS job_titles;
CREATE TABLE job_titles (
  job_title_id INT,
  job_title VARCHAR(125) NOT NULL,
  PRIMARY KEY (job_title_id)
);

-- =========================================
-- EMPLOYEE DIVISION
-- =========================================
DROP TABLE IF EXISTS employee_division;
CREATE TABLE employee_division (
  empid INT NOT NULL,
  div_ID INT NOT NULL,
  PRIMARY KEY (empid)
) COMMENT='links employee to a division';

-- =========================================
-- DIVISION
-- =========================================
DROP TABLE IF EXISTS division;
CREATE TABLE division (
  ID INT NOT NULL,
  Name VARCHAR(100) DEFAULT NULL,
  city VARCHAR(50) NOT NULL,
  addressLine1 VARCHAR(50) NOT NULL,
  addressLine2 VARCHAR(50) DEFAULT NULL,
  state VARCHAR(50) DEFAULT NULL,
  country VARCHAR(50) NOT NULL,
  postalCode VARCHAR(15) NOT NULL,
  PRIMARY KEY (ID)
) COMMENT='company divisions';

-- =========================================
-- ADDRESSES (Task 1.4.1)
-- =========================================
DROP TABLE IF EXISTS addresses;
CREATE TABLE addresses (
  street VARCHAR(100),
  cityID INT,
  stateID INT,
  zip VARCHAR(10),
  DOB DATE,
  phone VARCHAR(20),
  emergency_name VARCHAR(100),
  emergency_phone VARCHAR(20),
  empid INT
);

-- =========================================
-- CITIES (Task 1.4.2)
-- =========================================
DROP TABLE IF EXISTS cities;
CREATE TABLE cities (
  cityID INT AUTO_INCREMENT,
  city_name VARCHAR(25) NOT NULL,
  PRIMARY KEY (cityID)
);

-- =========================================
-- STATES (Task 1.4.2)
-- =========================================
DROP TABLE IF EXISTS states;
CREATE TABLE states (
  stateID INT AUTO_INCREMENT,
  state_code CHAR(2) NOT NULL,
  PRIMARY KEY (stateID)
);

-- =========================================
-- FOREIGN KEYS (Task 1.3)
-- =========================================

-- 1.3.1 employees -> employee_division
ALTER TABLE employee_division
ADD CONSTRAINT fk_emp_div
FOREIGN KEY (empid) REFERENCES employees(empid);

-- 1.3.2 employees -> payroll
ALTER TABLE payroll
ADD CONSTRAINT fk_emp_payroll
FOREIGN KEY (empid) REFERENCES employees(empid);

-- 1.3.3 employees -> addresses (via empid)
ALTER TABLE addresses
ADD CONSTRAINT fk_emp_address
FOREIGN KEY (empid) REFERENCES employees(empid);

-- 1.3.4 employees -> employee_job_titles
ALTER TABLE employee_job_titles
ADD CONSTRAINT fk_emp_job
FOREIGN KEY (empid) REFERENCES employees(empid);

-- 1.3.5 employee_division -> division
ALTER TABLE employee_division
ADD CONSTRAINT fk_division
FOREIGN KEY (div_ID) REFERENCES division(ID);

-- 1.3.6 employee_job_titles -> job_titles
ALTER TABLE employee_job_titles
ADD CONSTRAINT fk_job_title
FOREIGN KEY (job_title_id) REFERENCES job_titles(job_title_id);

-- =========================================
-- ADDITIONAL FOREIGN KEYS (3NF SUPPORT)
-- =========================================

ALTER TABLE addresses
ADD CONSTRAINT fk_city
FOREIGN KEY (cityID) REFERENCES cities(cityID);

ALTER TABLE addresses
ADD CONSTRAINT fk_state
FOREIGN KEY (stateID) REFERENCES states(stateID);

DELIMITER $$

DROP PROCEDURE IF EXISTS add_employee $$

CREATE PROCEDURE add_employee (
    IN p_fname VARCHAR(65),
    IN p_lname VARCHAR(65),
    IN p_email VARCHAR(65),
    IN p_hireDate DATE,
    IN p_salary DECIMAL(10,2),
    IN p_ssn VARCHAR(12),

    IN p_street VARCHAR(100),
    IN p_cityID INT,
    IN p_stateID INT,
    IN p_zip VARCHAR(10),
    IN p_DOB DATE,
    IN p_phone VARCHAR(20),
    IN p_emergency_name VARCHAR(100),
    IN p_emergency_phone VARCHAR(20),

    IN p_div_ID INT,
    IN p_job_title_id INT
)
BEGIN
    DECLARE new_empid INT;

    -- ERROR HANDLER
    DECLARE EXIT HANDLER FOR SQLEXCEPTION
    BEGIN
        ROLLBACK;
    END;

    START TRANSACTION;

    INSERT INTO employees (Fname, Lname, email, HireDate, Salary, SSN, addressID)
    VALUES (p_fname, p_lname, p_email, p_hireDate, p_salary, p_ssn, 0);

    SET new_empid = LAST_INSERT_ID();

    INSERT INTO addresses (street, cityID, stateID, zip, DOB, phone, emergency_name, emergency_phone, empid)
    VALUES (p_street, p_cityID, p_stateID, p_zip, p_DOB, p_phone, p_emergency_name, p_emergency_phone, new_empid);

    INSERT INTO employee_division (empid, div_ID)
    VALUES (new_empid, p_div_ID);

    INSERT INTO employee_job_titles (empid, job_title_id)
    VALUES (new_empid, p_job_title_id);

    COMMIT;
END $$

DELIMITER ;
